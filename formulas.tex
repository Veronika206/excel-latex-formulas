\section*{Формулы для расчетов}

% 1. Продольное напряжение
\begin{equation}
S = [(0{,}5 - \mu)\sigma_{\text{кц}} + \alpha E \Delta t]
\end{equation}

% 2. Осевой момент инерции
\begin{equation}
I = \frac{\pi}{64} \left( D_{\text{н}}^4 - D_{\text{вн}}^4 \right)
\end{equation}

% 3. Эквивалентная нагрузка (собственный вес)
\begin{equation}
q_{\text{м}} = n_{\text{с.в}} q_{\text{м}}^{\text{н}} = n_{\text{с.в}} \gamma_{\text{м}} F = n_{\text{с.в}} \gamma_{\text{м}} \frac{\pi}{4} \left( D_{\text{н}}^2 - D_{\text{вн}}^2 \right)
\end{equation}

% 4. Нагрузка от газа и от веса пригрузки
\begin{equation}
q_{\text{газ}}^{\text{н}} \approx 10^{-2} p D_{\text{вн}}^2
\end{equation}

\begin{equation}
q_{\text{пр}}^{\text{н}} = 10^{-4} \rho_{\text{н}} g \frac{\pi D_{\text{вн}}^2}{4}
\end{equation}

% 5. Суммарная нагрузка
\begin{equation}
q_{\text{тр}} = q_{\text{м}} + q_{\text{из}} + q_{\text{пр}}
\end{equation}

% 6. Давление грунта
\begin{equation}
p_{\text{гр}} = \frac{2 n_{\text{гр}} \gamma_{\text{гр}} D_{\text{н}} \left[ \left( h_{\text{o}} + \frac{D_{\text{н}}}{8} \right) + \left( h_{\text{o}} + \frac{D_{\text{н}}}{2} \right) \operatorname{tg}^2 \left( 45^{\circ} - \frac{\varphi_{\text{гр}}}{2} \right) \right] + q_{\text{тр}}}{\pi D_{\text{н}}}
\end{equation}

% 7. Сопротивление грунта
\begin{equation}
p_{\text{o}} = p D_{\text{н}} \tau_{\text{пр}}
\end{equation}

% 8. Собственный вес (с учетом грунта)
\begin{equation}
q_{\text{в}} = n_{\text{гр}} \gamma_{\text{гр}} D_{\text{н}} (h_{\text{o}} + D_{\text{н}}/2 - \pi D_{\text{н}}/8) + q_{\text{тр}}
\end{equation}

% 9. Критическое усилие (формула Эйлера)
\begin{equation}
N_{\text{кр}} = 4{,}09 \sqrt[11]{p_{\text{o}}^2 q_{\text{в}}^4 F^2 E^5 I^3}
\end{equation}

% 10. Условие устойчивости
\begin{equation}
S \le \frac{m}{1{,}1} \cdot N_{\text{кр}}
\end{equation}

% 11. Критическая длина
\begin{equation}
\lambda_{\text{кр}} = 2\pi \sqrt[4]{EI / k_0 D_{\text{н}}}
\end{equation}

% 12. Критическое усилие (сокращенная формула)
\begin{equation}
N_{\text{кр}} = 2 \sqrt{k_0 D_{\text{н}} EI}
\end{equation}

% 13. Критическое усилие (через qв)
\begin{equation}
N_{\text{кр}} = 0{,}375 q_{\text{в}} \rho
\end{equation}

% 14. Геометрические условия и коэффициенты
\begin{equation}
1 : 1{,}25, \quad \alpha_{\text{н}} \le 39^{\circ}
\end{equation}

\begin{equation}
\rho_{\text{пр}} = n_{\text{гр}} \gamma_{\text{гр}}
\end{equation}

\begin{equation}
E_1 = \frac{n_{\text{гр}} \rho_{\text{гр}} h_{\text{гр}}^2}{2} \operatorname{tg}^2(45^{\circ} - \dots)
\end{equation}

\begin{equation}
E_{\text{акт}} = - \frac{n_{\text{гр}} \rho_{\text{гр}} h_{\text{гр}}^2}{2} \operatorname{tg}^2(45^{\circ} - \dots)
\end{equation}

% 15. Ширина траншеи
\begin{equation}
a = b + H/\operatorname{tg} \alpha_{\text{н}}
\end{equation}

% 16. Стрела прогиба
\begin{equation}
f_{\text{кр}} = \rho - \sqrt{\rho^2 - l_{\text{кр}}^2 / 4}
\end{equation}

% 17. Коэффициент
\begin{equation}
k_{\alpha} = \sqrt{\sin^2 \frac{\alpha_{\text{в}}}{2} + \cos^4 \frac{\alpha_{\text{в}}}{2}}
\end{equation}
